Integrasi Google Sheets

Sinkronkan data keuangan lokal Anda ke Google Spreadsheet

Belum pernah disinkronkan

Sinkronisasi Otomatis (Real-time)

Otomatis kirim perubahan ke Google Sheets saat menambah, mengedit, atau menghapus transaksi.

Panduan & Kode Google Apps Script

  1. Buka Google Sheets baru, lalu klik menu Ekstensi (Extensions) > Apps Script.
  2. Hapus seluruh kode yang ada, lalu tempel (paste) kode di bawah ini dan klik Simpan.
  3. Klik Deploy > New deployment, pilih tipe Web app.
  4. Atur Execute as: Me dan Who has access: Anyone, lalu klik Deploy.
  5. Salin Web app URL yang muncul dan tempelkan pada kolom di atas.
const SHEET_NAME = "Transaksi";

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(["ID", "Tanggal", "Jenis", "Kategori", "Dompet", "Nominal", "Catatan"]);
    sheet.getRange("A1:G1").setFontWeight("bold");
  }
  return sheet;
}

function doGet(e) {
  const sheet = getOrCreateSheet();
  const rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify({ status: "ok", transactions: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  const transactions = rows.slice(1).filter(r => r[0]).map(r => ({
    id: String(r[0]),
    date: r[1] instanceof Date ? Utilities.formatDate(r[1], Session.getScriptTimeZone(), "yyyy-MM-dd") : String(r[1]).slice(0, 10),
    type: r[2] === "Pemasukan" || r[2] === "income" ? "income" : "expense",
    category: String(r[3]),
    wallet: String(r[4]),
    amount: Number(r[5]) || 0,
    note: String(r[6] || "")
  }));
  return ContentService.createTextOutput(JSON.stringify({ status: "ok", transactions }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = getOrCreateSheet();
  const payload = JSON.parse(e.postData.contents);
  const action = payload.action || "syncAll";

  if (action === "add" && payload.transaction) {
    const t = payload.transaction;
    sheet.appendRow([t.id, t.date, t.type === "income" ? "Pemasukan" : "Pengeluaran", t.category, t.wallet, Number(t.amount), t.note]);
  } else if (action === "update" && payload.transaction) {
    const t = payload.transaction;
    const rows = sheet.getDataRange().getValues();
    let updated = false;
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(t.id)) {
        sheet.getRange(i + 1, 1, 1, 7).setValues([[t.id, t.date, t.type === "income" ? "Pemasukan" : "Pengeluaran", t.category, t.wallet, Number(t.amount), t.note]]);
        updated = true;
        break;
      }
    }
    if (!updated) {
      sheet.appendRow([t.id, t.date, t.type === "income" ? "Pemasukan" : "Pengeluaran", t.category, t.wallet, Number(t.amount), t.note]);
    }
  } else if (action === "delete" && payload.id) {
    const rows = sheet.getDataRange().getValues();
    for (let i = rows.length - 1; i >= 1; i--) {
      if (String(rows[i][0]) === String(payload.id)) {
        sheet.deleteRow(i + 1);
        break;
      }
    }
  } else if (action === "syncAll" && Array.isArray(payload.transactions)) {
    sheet.clearContents();
    sheet.appendRow(["ID", "Tanggal", "Jenis", "Kategori", "Dompet", "Nominal", "Catatan"]);
    sheet.getRange("A1:G1").setFontWeight("bold");
    if (payload.transactions.length > 0) {
      const dataRows = payload.transactions.map(t => [
        t.id, t.date, t.type === "income" ? "Pemasukan" : "Pengeluaran", t.category, t.wallet, Number(t.amount), t.note
      ]);
      sheet.getRange(2, 1, dataRows.length, 7).setValues(dataRows);
    }
  }

  return ContentService.createTextOutput(JSON.stringify({ status: "ok", message: "Berhasil disinkronkan" }))
    .setMimeType(ContentService.MimeType.JSON);
}

Instal Aplikasi

Di iPhone / iPad (Safari), instal KEUANGANKU dengan langkah berikut:

  1. Ketuk ikon Bagikan (Share) di bilah bawah Safari.
  2. Gulir ke bawah, lalu pilih "Tambahkan ke Layar Utama" (Add to Home Screen).
  3. Ketuk Tambah. Aplikasi akan muncul di layar utama Anda.

Aplikasi akan terbuka tanpa bilah alamat browser, layaknya aplikasi native, dan data keuangan Anda tetap tersimpan aman di perangkat.
